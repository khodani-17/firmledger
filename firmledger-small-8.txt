export async function reviewItem(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const reviewId = String(
    formData.get("reviewId") || "",
  );

  const status = String(
    formData.get("status") || "REVIEWED",
  );

  if (
    ![
      "REVIEWED",
      "FLAGGED",
      "CORRECTED",
    ].includes(status)
  ) {
    throw new Error(
      "Invalid review status.",
    );
  }

  const review =
    await prisma.accountantReview.findFirst({
      where: {
        id: reviewId,
        firmId: user.firmId,
      },
    });

  if (!review) {
    throw new Error(
      "Review item not found.",
    );
  }

  await prisma.accountantReview.update({
    where: {
      id: review.id,
    },
    data: {
      status: status as any,
      reviewerId: user.id,
      reviewedAt: new Date(),
    },
  });

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "REVIEW",
    entity: "AccountantReview",
    entityId: review.id,
    metadata: {
      status,
    },
  });

  revalidatePath("/accounting");
}

export async function addBankTransaction(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const bankAccountId = String(
    formData.get("bankAccountId") || "",
  );

  const description = String(
    formData.get("description") || "",
  ).trim();

  const reference =
    String(
      formData.get("reference") || "",
    ).trim() || null;

  const amount = money(
    String(formData.get("amount") || "0"),
  );

  const type = String(
    formData.get("type") || "CREDIT",
  );

  if (
    !bankAccountId ||
    !description ||
    amount <= 0 ||
    !["CREDIT", "DEBIT"].includes(type)
  ) {
    throw new Error(
      "Bank, description, amount and transaction type are required.",
    );
  }

  const bank =
    await prisma.bankAccount.findFirst({
      where: {
        id: bankAccountId,
        firmId: user.firmId,
        active: true,
      },
    });

  if (!bank) {
    throw new Error(
      "Bank account not found.",
    );
  }

  const tx =
    await prisma.bankTransaction.create({
      data: {
        firmId: user.firmId,
        bankAccountId,
        transactionDate: new Date(),
        description,
        reference,
        amount,
        type: type as any,
        matched: false,
      },
    });

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "CREATE",
    entity: "BankTransaction",
    entityId: tx.id,
  });

  revalidatePath("/banking");
  revalidatePath(
    "/banking/reconciliation",
  );

  redirect("/banking/reconciliation");
}

export async function reconcileBankTransaction(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const bankTransactionId = String(
    formData.get("bankTransactionId") || "",
  );

  const paymentId = String(
    formData.get("paymentId") || "",
  );

  const transaction =
    await prisma.bankTransaction.findFirst({
      where: {
        id: bankTransactionId,
        bankAccount: {
          firmId: user.firmId,
        },
        matched: false,
      },
    });

  if (!transaction) {
    throw new Error(
      "Unmatched bank transaction not found.",
    );
  }

  if (!paymentId) {
    throw new Error(
      "Select a payment to reconcile.",
    );
  }

  const payment =
    await prisma.payment.findFirst({
      where: {
        id: paymentId,
        firmId: user.firmId,
        status: "COMPLETED",
      },
      include: {
        bankTransactions: true,
      },
    });

  if (!payment) {
    throw new Error("Payment not found.");
  }

  if (transaction.type !== "CREDIT") {
    throw new Error(
      "This simple reconciliation flow only matches incoming client receipts.",
    );
  }

  if (
    Math.abs(
      Number(transaction.amount) -
        Number(payment.amount),
    ) > 0.001
  ) {
    throw new Error(
      "Bank amount and payment amount do not match.",
    );
  }

  if (payment.bankTransactions.length) {
    throw new Error(
      "Payment is already linked to a bank transaction.",
    );
  }

  await prisma.$transaction(
    async (tx) => {
      await tx.bankTransaction.update({
        where: {
          id: transaction.id,
        },
        data: {
          matched: true,
          paymentId: payment.id,
        },
      });
    },
  );

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "RECONCILE",
    entity: "BankTransaction",
    entityId: transaction.id,
    metadata: {
      paymentId,
    },
  });

  revalidatePath("/banking");
  revalidatePath(
    "/banking/reconciliation",
  );

  redirect("/banking/reconciliation");
}
