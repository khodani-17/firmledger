export async function createCreditNote(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const clientId = String(
    formData.get("clientId") || "",
  );

  const invoiceId =
    String(
      formData.get("invoiceId") || "",
    ) || null;

  const reason = String(
    formData.get("reason") || "",
  ).trim();

  const subtotal = money(
    String(
      formData.get("subtotal") || "0",
    ),
  );

  const tax = money(
    String(
      formData.get("taxAmount") || "0",
    ),
  );

  if (
    !clientId ||
    !reason ||
    subtotal <= 0
  ) {
    throw new Error(
      "Client, reason and amount are required.",
    );
  }

  const count =
    await prisma.creditNote.count({
      where: {
        firmId: user.firmId,
      },
    });

  const total =
    Math.round(
      (subtotal + tax) * 100,
    ) / 100;

  const note =
    await prisma.$transaction(
      async (tx) => {
        if (invoiceId) {
          const inv =
            await tx.invoice.findFirst({
              where: {
                id: invoiceId,
                firmId: user.firmId,
                clientId,
              },
            });

          if (!inv) {
            throw new Error(
              "Invoice does not belong to the client.",
            );
          }
        }

        const created =
          await tx.creditNote.create({
            data: {
              firmId: user.firmId,
              clientId,
              invoiceId,
              creditNumber: `CN-${String(
                count + 1,
              ).padStart(4, "0")}`,
              issueDate: new Date(),
              reason,
              subtotal,
              taxAmount: tax,
              total,
            },
          });

        if (invoiceId) {
          const inv =
            await tx.invoice.findFirst({
              where: {
                id: invoiceId,
                firmId: user.firmId,
                clientId,
              },
            });

          if (inv) {
            const balance =
              Math.max(
                0,
                Number(inv.balanceDue) -
                  total,
              );

            await tx.invoice.update({
              where: {
                id: inv.id,
              },
              data: {
                balanceDue: balance,
                status:
                  balance === 0
                    ? "PAID"
                    : inv.status,
              },
            });
          }
        }

        return created;
      },
    );

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "CREATE",
    entity: "CreditNote",
    entityId: note.id,
    metadata: {
      total,
    },
  });

  revalidatePath("/credit-notes");
  revalidatePath("/billing");

  redirect("/credit-notes");
}

export async function createSupplierPayment(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const supplierId = String(
    formData.get("supplierId") || "",
  );

  const purchaseInvoiceId =
    String(
      formData.get("purchaseInvoiceId") ||
        "",
    ) || null;

  const bankAccountId = String(
    formData.get("bankAccountId") || "",
  );

  const reference = String(
    formData.get("reference") || "",
  ).trim();

  const amount = money(
    String(
      formData.get("amount") || "0",
    ),
  );

  if (
    !supplierId ||
    !bankAccountId ||
    !reference ||
    amount <= 0
  ) {
    throw new Error(
      "Supplier, bank account, reference and amount are required.",
    );
  }

  const supplier =
    await prisma.supplier.findFirst({
      where: {
        id: supplierId,
        firmId: user.firmId,
      },
    });

  if (!supplier) {
    throw new Error(
      "Supplier not found.",
    );
  }

  const bank =
    await prisma.bankAccount.findFirst({
      where: {
        id: bankAccountId,
        firmId: user.firmId,
        active: true,
        isTrust: false,
      },
    });

  if (!bank) {
    throw new Error(
      "Business bank account not found.",
    );
  }

  const payment =
    await prisma.$transaction(
      async (tx) => {
        let invoice: any = null;

        if (purchaseInvoiceId) {
          invoice =
            await tx.purchaseInvoice.findFirst({
              where: {
                id: purchaseInvoiceId,
                firmId: user.firmId,
                supplierId,
              },
            });

          if (!invoice) {
            throw new Error(
              "Purchase invoice not found.",
            );
          }

          if (
            amount >
            Number(
              invoice.balanceDue,
            ) + 0.001
          ) {
            throw new Error(
              `Payment exceeds payable balance of R${Number(
                invoice.balanceDue,
              ).toFixed(2)}.`,
            );
          }
        }

        const paymentDate =
          new Date();

        const created =
          await tx.supplierPayment.create({
            data: {
              firmId: user.firmId,
              supplierId,
              purchaseInvoiceId,
              bankAccountId,
              paymentDate,
              reference,
              amount,
            },
          });

        if (invoice) {
          const balance =
            Math.round(
              (Number(
                invoice.balanceDue,
              ) - amount) *
                100,
            ) / 100;

          await tx.purchaseInvoice.update(
            {
              where: {
                id: invoice.id,
              },
              data: {
                amountPaid: {
                  increment: amount,
                },
                balanceDue: balance,
                status:
                  balance === 0
                    ? "PAID"
                    : "PARTIALLY_PAID",
              },
            },
          );
        }

        await postSupplierPayment(tx, {
          firmId: user.firmId,
          id: created.id,
          reference,
          date: paymentDate,
          amount,
        });

        await tx.bankTransaction.create({
          data: {
            firmId: user.firmId,
            bankAccountId,
            transactionDate:
              paymentDate,
            description: `Supplier payment - ${supplier.name}`,
            reference,
            amount,
            type: "DEBIT",
            matched: true,
          },
        });

        await tx.accountantReview.create({
          data: {
            firmId: user.firmId,
            title: `Review supplier payment ${reference}`,
            status: "PENDING",
            notes: `R${amount.toFixed(
              2,
            )} paid to ${supplier.name}.`,
          },
        });

        return created;
      },
    );

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "CREATE_AND_POST",
    entity: "SupplierPayment",
    entityId: payment.id,
    metadata: {
      amount,
      reference,
    },
  });

  revalidatePath("/purchases");
  revalidatePath("/banking");
  revalidatePath("/accounting");

  redirect("/purchases");
}
