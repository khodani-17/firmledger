# FirmLedger by Manzima

FirmLedger is a South African law-firm financial management and bookkeeping platform designed for small and growing practices. It keeps lawyer-facing workflows simple while giving finance/accounting users a controlled double-entry accounting workspace.

## Current platform scope

- Multi-tenant firms and role-based access
- Clients and matters
- Client billing and VAT
- Client payments and invoice allocation
- Double-entry journals and trial balance
- Business bank accounts and reconciliation
- Expenses and disbursements
- Suppliers and purchase invoices
- Accounts payable and supplier payments
- Accountant review queue and audit trail
- Financial periods and month-end close controls
- VAT/tax working-paper preparation from posted journals
- Recurring billing schedules
- Credit notes
- Dedicated trust accounts, client/matter trust ledgers and trust reconciliation records
- Management financial reports

## Accounting controls

System-generated postings are balanced before they are saved. Business banking is separated from trust accounting. Trust transactions are held in their own subsystem and require client allocation. Closed periods and accountant review are intended to support month-end control.

FirmLedger does not claim to replace professional accounting or statutory responsibility. VAT/tax outputs are prepared as working figures and should be reviewed and submitted by the responsible professional/firm process.

## Demo credentials

- owner@example.com / ChangeMe123!
- finance@example.com / ChangeMe123!
- accountant@example.com / ChangeMe123!
- lawyer@example.com / ChangeMe123!

Change or remove demo credentials before production.

## Stack

- Next.js 16
- React 19
- TypeScript
- Prisma 7
- PostgreSQL / Neon

## Setup

1. Copy `.env.example` to `.env`.
2. Set `DATABASE_URL` and a strong `AUTH_SECRET`.
3. Install dependencies with `npm install`.
4. Generate Prisma client: `npm run db:generate`.
5. Push the schema: `npm run db:push`.
6. Seed development data: `npm run db:seed`.
7. Run: `npm run dev`.

## Important development note

The source was expanded to the full accounting workflow, but dependency installation/build verification still needs to be completed in an environment where npm package installation can finish successfully. The project intentionally excludes `node_modules` and `.next` from the package.

  );

  const subtotalRaw = String(

    formData.get("taxRate") || "0",

   * VAT calculation:
   *
   * Example:
   * subtotal = 10,000
   * taxRate = 15
   *
   * taxAmount = 10,000 × 15 / 100 = 1,500
   * total = 10,000 + 1,500 = 11,500
   */
  const taxAmount =
    Math.round(subtotal * taxRate) / 100;

  const total =
    Math.round(
      (subtotal + taxAmount) * 100,
    ) / 100;

  console.log("INVOICE CALCULATION:", {
    subtotalRaw,
    taxRateRaw,
    subtotal,
    taxRate,
    taxAmount,
    total,
  });

  if (
    !clientId ||
    !description ||
    subtotal <= 0
  ) {
    throw new Error(
      "Client, description and amount are required.",
    );
  }

  if (taxRate < 0 || taxRate > 100) {
    throw new Error(
      "VAT rate must be between 0% and 100%.",
    );
  }

  const client = await prisma.client.findFirst({
    where: {
      id: clientId,
      firmId: user.firmId,
    },
  });

  if (!client) {
    throw new Error("Client not found.");
  }

  if (matterId) {
    const matter = await prisma.matter.findFirst({
      where: {
        id: matterId,
        firmId: user.firmId,
        clientId,
      },
    });

    if (!matter) {
      throw new Error(
        "Matter does not belong to this client.",
      );
    }
  }

  const count = await prisma.invoice.count({
    where: {
      firmId: user.firmId,
    },
  });

  const invoiceNumber = `INV-${String(
    count + 1,
  ).padStart(4, "0")}`;

  const issueDate = new Date();

  const dueDate = new Date();
  dueDate.setDate(
    dueDate.getDate() + 30,
  );

  const invoice = await prisma.$transaction(
    async (tx) => {
      const created =
        await tx.invoice.create({
          data: {
            firmId: user.firmId,
            clientId,
            matterId,
            invoiceNumber,
            issueDate,
            dueDate,
            status: "SENT",
            subtotal,
            taxAmount,
            total,
            balanceDue: total,

            items: {
              create: [
                {
                  description,
                  quantity: 1,
                  unitPrice: subtotal,
                  taxRate,
                  lineTotal: subtotal,
                },
              ],
            },
          },
        });

      await postInvoice(tx, {
        firmId: user.firmId,

      });

    },
  );

export async function createPayment(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const invoiceId = String(
    formData.get("invoiceId") || "",
  );

  const bankAccountId = String(
    formData.get("bankAccountId") || "",
  );

  const amount = money(
    String(formData.get("amount") || "0"),
  );

  const reference = String(
    formData.get("reference") || "",
  ).trim();

  const method = String(
    formData.get("method") || "EFT",
  ).trim();

  if (
    !invoiceId ||
    !bankAccountId ||
    amount <= 0 ||
    !reference
  ) {
    throw new Error(
      "Invoice, bank account, amount and reference are required.",
    );
  }

  const payment =
    await prisma.$transaction(async (tx) => {
      const invoice =
        await tx.invoice.findFirst({
          where: {
            id: invoiceId,
            firmId: user.firmId,
          },
        });

      if (!invoice) {
        throw new Error("Invoice not found.");
      }

      const bank =
        await tx.bankAccount.findFirst({
          where: {
            id: bankAccountId,
            firmId: user.firmId,
            active: true,
          },
        });

      if (!bank) {
        throw new Error(
          "Bank account not found.",
        );
      }

      const balance = Number(
        invoice.balanceDue,
      );

      if (amount > balance + 0.0001) {
        throw new Error(
          `Payment exceeds the invoice balance of R${balance.toFixed(
            2,
          )}.`,
        );
      }

      const paymentDate = new Date();

      const created =
        await tx.payment.create({
          data: {
            firmId: user.firmId,
            clientId: invoice.clientId,
            matterId: invoice.matterId,
            invoiceId,
            reference,
            paymentDate,
            amount,
            status: "COMPLETED",
            method,
          },
        });

      const newPaid =
        Math.round(
          (Number(invoice.amountPaid) +
            amount) *
            100,
        ) / 100;

      const newBalance =
        Math.round(
          (balance - amount) *
            100,
        ) / 100;

      await tx.invoice.update({
        where: {
          id: invoice.id,
        },
        data: {
          amountPaid: newPaid,
          balanceDue: newBalance,
          status:
            newBalance === 0
              ? "PAID"
              : "PARTIALLY_PAID",
        },
      });

      await postPayment(tx, {
        firmId: user.firmId,
        paymentId: created.id,
        reference,
        entryDate: paymentDate,
        amount,
      });

      await tx.bankTransaction.create({
        data: {
          firmId: user.firmId,
          bankAccountId,
          paymentId: created.id,
          transactionDate: paymentDate,
          description: `Client payment - ${invoice.invoiceNumber}`,
          reference,
          amount,
          type: "CREDIT",
          matched: true,
        },
      });

      await tx.accountantReview.create({
        data: {
          firmId: user.firmId,
          title: `Review payment ${reference}`,
          status: "PENDING",
          notes: `R${amount.toFixed(
            2,
          )} allocated to ${invoice.invoiceNumber}.`,
        },
      });

      return created;
    });

    firmId: user.firmId,
    userId: user.id,

    entity: "Payment",
    entityId: payment.id,
    metadata: {
      reference,

export async function reconcileBankTransaction(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const bankTransactionId = String(
    formData.get("bankTransactionId") || "",
  );

  const paymentId = String(
    formData.get("paymentId") || "",
  );

  const transaction =
    await prisma.bankTransaction.findFirst({
      where: {
        id: bankTransactionId,
        bankAccount: {
          firmId: user.firmId,
        },
        matched: false,
      },
    });

  if (!transaction) {
    throw new Error(
      "Unmatched bank transaction not found.",
    );
  }

  if (!paymentId) {
    throw new Error(
      "Select a payment to reconcile.",
    );
  }

  const payment =
    await prisma.payment.findFirst({
      where: {
        id: paymentId,
        firmId: user.firmId,
        status: "COMPLETED",
      },
      include: {
        bankTransactions: true,
      },
    });

  if (!payment) {
    throw new Error("Payment not found.");
  }

  if (transaction.type !== "CREDIT") {
    throw new Error(
      "This simple reconciliation flow only matches incoming client receipts.",
    );
  }

  if (
    Math.abs(
      Number(transaction.amount) -
        Number(payment.amount),
    ) > 0.001
  ) {
    throw new Error(
      "Bank amount and payment amount do not match.",
    );
  }

  if (payment.bankTransactions.length) {
    throw new Error(
      "Payment is already linked to a bank transaction.",
    );
  }

  await prisma.$transaction(
    async (tx) => {
      await tx.bankTransaction.update({
        where: {
          id: transaction.id,
        },
        data: {
          matched: true,
          paymentId: payment.id,
        },
      });
    },
  );

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "RECONCILE",
    entity: "BankTransaction",
    entityId: transaction.id,
    metadata: {
      paymentId,
    },
  });

  revalidatePath("/banking");
  revalidatePath(
    "/banking/reconciliation",
  );

  redirect("/banking/reconciliation");
}

export async function createPurchaseInvoice(
  formData: FormData,

    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const supplierId = String(
    formData.get("supplierId") || "",
  );

    formData.get("invoiceNumber") || "",
  ).trim();

  const description = String(
    formData.get("description") || "",
  ).trim();

    String(
      formData.get("subtotal") || "0",
    ),
  );

  const tax = money(
    String(
      formData.get("taxAmount") || "0",
    ),
  );

  if (
    !supplierId ||
    !invoiceNumber ||
    !description ||
    net <= 0
  ) {
    throw new Error(
      "Supplier, invoice number, description and amount are required.",
    );
  }

    await prisma.supplier.findFirst({

        id: supplierId,

    await prisma.$transaction(

          await tx.purchaseInvoice.create({

          firmId: user.firmId,
          id: created.id,
          description: `${supplier.name} ${invoiceNumber}`,
          date,
          net,
          tax,

          data: {
            firmId: user.firmId,
            title: `Review supplier invoice ${invoiceNumber}`,
            status: "PENDING",
            notes: `R${total.toFixed(
              2,

export async function createTrustTransaction(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const trustAccountId = String(
    formData.get("trustAccountId") || "",
  );

  const clientId = String(
    formData.get("clientId") || "",
  );

  const matterId =
    String(
      formData.get("matterId") || "",
    ) || null;

  const amount = money(
    String(
      formData.get("amount") || "0",
    ),
  );

  const type = String(
    formData.get("type") || "RECEIPT",
  );

  const description = String(
    formData.get("description") || "",
  ).trim();

  const reference =
    String(
      formData.get("reference") || "",
    ).trim() || null;

  if (
    !trustAccountId ||
    !clientId ||
    !description ||
    amount <= 0 ||
    ![
      "RECEIPT",
      "PAYMENT",
      "TRANSFER",
      "INTEREST",
      "FEE",
      "REVERSAL",
    ].includes(type)
  ) {
    throw new Error(
      "Trust account, client, description and valid amount are required.",
    );
  }

  const ta =
    await prisma.trustAccount.findFirst({
      where: {
        id: trustAccountId,
        firmId: user.firmId,
        active: true,
      },
    });

  if (!ta) {
    throw new Error(
      "Trust account not found.",
    );
  }

  const client =
    await prisma.client.findFirst({
      where: {
        id: clientId,
        firmId: user.firmId,
      },
    });

  if (!client) {
    throw new Error(
      "Client not found.",
    );
  }

  if (matterId) {
    const matter =
      await prisma.matter.findFirst({
        where: {
          id: matterId,
          firmId: user.firmId,
          clientId,
        },
      });

    if (!matter) {
      throw new Error(
        "Matter does not belong to the client.",
      );
    }
  }

  const current =
    await prisma.trustTransaction.findMany({
      where: {
        trustAccountId,
        status: "POSTED",
      },
      select: {
        amount: true,
        type: true,
      },
    });

  const balance =
    current.reduce(
      (s, t) =>
        s +
        (t.type === "RECEIPT" ||
        t.type === "INTEREST"
          ? Number(t.amount)
          : -Number(t.amount)),
      0,
    );

  if (
    [
      "PAYMENT",
      "TRANSFER",
      "FEE",
    ].includes(type) &&
    amount > balance + 0.001
  ) {
    throw new Error(
      `Insufficient trust funds. Available balance: R${balance.toFixed(
        2,
      )}.`,
    );
  }

  const tx =
    await prisma.trustTransaction.create({
      data: {
        firmId: user.firmId,
        trustAccountId,
        clientId,
        matterId,
        type: type as any,
        status: "POSTED",
        transactionDate: new Date(),
        reference,
        description,
        amount,
        createdById: user.id,
      },
    });

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "POST",
    entity: "TrustTransaction",
    entityId: tx.id,
    metadata: {
      amount,
      type,
      clientId,
      matterId,
    },
  });

  revalidatePath("/trust");
  redirect("/trust");
}

export async function createTaxReturn(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const start = new Date(
    String(
      formData.get("periodStart") || "",
    ),
  );

  const end = new Date(
    String(
      formData.get("periodEnd") || "",
    ),
  );

  if (
    isNaN(start.getTime()) ||
    isNaN(end.getTime()) ||
    start >= end
  ) {
    throw new Error(
      "Valid tax period required.",
    );
  }

  const lines =
    await prisma.journalLine.findMany({
      where: {
        journalEntry: {
          firmId: user.firmId,
          entryDate: {
            gte: start,
            lte: end,
          },
        },
      },
      include: {
        ledgerAccount: true,
      },
    });

  const output = lines
    .filter(
      (l) =>
        l.ledgerAccount.code === "2000",
    )
    .reduce(
      (s, l) =>
        s +
        Number(l.credit) -
        Number(l.debit),
      0,
    );

  const input = lines
    .filter(
      (l) =>
        l.ledgerAccount.code === "2100",
    )
    .reduce(
      (s, l) =>
        s +
        Number(l.debit) -
        Number(l.credit),
      0,
    );

  const ret =
    await prisma.taxReturn.create({
      data: {
        firmId: user.firmId,
        returnType: String(
          formData.get("returnType") ||
            "VAT201",
        ),
        periodStart: start,
        periodEnd: end,
        outputVat: output,
        inputVat: input,
        netVat: output - input,
        status: "PREPARED",
        preparedBy: user.id,
        preparedAt: new Date(),
      },
    });

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "PREPARE",
    entity: "TaxReturn",
    entityId: ret.id,
    metadata: {
      output,
      input,
      net: output - input,
    },
  });

  revalidatePath("/tax");
  redirect("/tax");
}

export async function createCreditNote(
  formData: FormData,
) {
  const user = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const clientId = String(
    formData.get("clientId") || "",
  );

  const invoiceId =
    String(
      formData.get("invoiceId") || "",
    ) || null;

  const reason = String(
    formData.get("reason") || "",
  ).trim();

  const subtotal = money(
    String(
      formData.get("subtotal") || "0",
    ),
  );

  const tax = money(
    String(
      formData.get("taxAmount") || "0",
    ),
  );

  if (
    !clientId ||
    !reason ||
    subtotal <= 0
  ) {
    throw new Error(
      "Client, reason and amount are required.",
    );
  }

  const count =
    await prisma.creditNote.count({
      where: {
        firmId: user.firmId,
      },
    });

  const total =
    Math.round(
      (subtotal + tax) * 100,
    ) / 100;

  const note =
    await prisma.$transaction(
      async (tx) => {
        if (invoiceId) {
          const inv =
            await tx.invoice.findFirst({
              where: {
                id: invoiceId,
                firmId: user.firmId,
                clientId,
              },
            });

          if (!inv) {
            throw new Error(
              "Invoice does not belong to the client.",
            );
          }
        }

        const created =
          await tx.creditNote.create({
            data: {
              firmId: user.firmId,
              clientId,
              invoiceId,
              creditNumber: `CN-${String(
                count + 1,
              ).padStart(4, "0")}`,
              issueDate: new Date(),
              reason,
              subtotal,
              taxAmount: tax,
              total,
            },
          });

        if (invoiceId) {
          const inv =
            await tx.invoice.findFirst({
              where: {
                id: invoiceId,
                firmId: user.firmId,
                clientId,
              },
            });

          if (inv) {
            const balance =
              Math.max(
                0,
                Number(inv.balanceDue) -
                  total,
              );

            await tx.invoice.update({
              where: {
                id: inv.id,
              },
              data: {
                balanceDue: balance,
                status:
                  balance === 0
                    ? "PAID"
                    : inv.status,
              },
            });
          }
        }

        return created;
      },
    );

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "CREATE",
    entity: "CreditNote",
    entityId: note.id,
    metadata: {
      total,

            ) + 0.001
          ) {
            throw new Error(
              `Payment exceeds payable balance of R${Number(
                invoice.balanceDue,
              ).toFixed(2)}.`,
            );
          }
        }

        const paymentDate =
          new Date();

        const created =
          await tx.supplierPayment.create({
            data: {
              firmId: user.firmId,
              supplierId,
              purchaseInvoiceId,
              bankAccountId,
              paymentDate,
              reference,
              amount,
            },
          });

        if (invoice) {
          const balance =
            Math.round(
              (Number(
                invoice.balanceDue,
              ) - amount) *
                100,
            ) / 100;

          await tx.purchaseInvoice.update(
            {
              where: {
                id: invoice.id,
              },
              data: {
                amountPaid: {
                  increment: amount,
                },
                balanceDue: balance,
                status:
                  balance === 0
                    ? "PAID"
                    : "PARTIALLY_PAID",
              },
            },
          );
        }

        await postSupplierPayment(tx, {
          firmId: user.firmId,
          id: created.id,
          reference,
          date: paymentDate,
          amount,
        });

        await tx.bankTransaction.create({
          data: {
            firmId: user.firmId,
            bankAccountId,
            transactionDate:
              paymentDate,
            description: `Supplier payment - ${supplier.name}`,
            reference,
            amount,
            type: "DEBIT",
            matched: true,
          },
        });

        await tx.accountantReview.create({
          data: {
            firmId: user.firmId,
            title: `Review supplier payment ${reference}`,
            status: "PENDING",
            notes: `R${amount.toFixed(
              2,
            )} paid to ${supplier.name}.`,
          },
        });

        return created;
      },
    );

  await audit({
    firmId: user.firmId,
    userId: user.id,
    action: "CREATE_AND_POST",
    entity: "SupplierPayment",
    entityId: payment.id,
    metadata: {
      amount,
      reference,
    },
  });

  revalidatePath("/purchases");
  revalidatePath("/banking");
  revalidatePath("/accounting");

  redirect("/purchases");
}

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import {
  makeSession,
  verifyPassword,
  SESSION_COOKIE,
  SESSION_TTL_SECONDS,
} from "@/lib/auth";

export async function POST(request: Request) {
  try {
    const body = await request.json();

    const email = String(body.email || "")
      .trim()
      .toLowerCase();

    const password = String(body.password || "");

    if (!email || !password) {
      return NextResponse.json(
        {
          error: "Email and password are required.",
        },
        { status: 400 }
      );
    }

    const user = await prisma.firmUser.findFirst({
      where: {
        email,
        active: true,
      },
    });

    if (!user) {
      return NextResponse.json(
        {
          error: "Invalid email or password.",
        },
        { status: 401 }
      );
    }

    const validPassword = await verifyPassword(
      password,
      user.passwordHash
    );

    if (!validPassword) {
      return NextResponse.json(
        {
          error: "Invalid email or password.",
        },
        { status: 401 }
      );
    }

    const response = NextResponse.json({
      ok: true,
    });

    response.cookies.set({
      name: SESSION_COOKIE,
      value: makeSession(user.id),
      httpOnly: true,
      sameSite: "lax",
      secure: process.env.NODE_ENV === "production",
      path: "/",
      maxAge: SESSION_TTL_SECONDS,
    });

    return response;
  } catch (error) {
    console.error("Login error:", error);

    return NextResponse.json(
      {
        error: "Unable to sign in. Please try again.",
      },
      { status: 500 }
    );
  }
}

    const email = String(body.email || "")
      .trim()
      .toLowerCase();
    const password = String(body.password || "");
    const confirmPassword = String(body.confirmPassword || "");

    if (!firmName) {
      return NextResponse.json(
        {

      return NextResponse.json(

        },
        {

        },
      );

    if (password !== confirmPassword) {

        {
          status: 400,

      );
    }

    const existingFirm = await prisma.firm.findFirst({

import Link from "next/link";
import { FormEvent, useState } from "react";
import { useRouter } from "next/navigation";

export default function LoginPage() {
  const router = useRouter();

  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    setError("");
    setLoading(true);

    try {
      const response = await fetch("/api/auth/login", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          email,
          password,
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        setError(data.error || "Invalid email or password.");
        setLoading(false);
        return;
      }

      router.replace("/");
      router.refresh();
    } catch {
      setError("Unable to sign in. Please try again.");
      setLoading(false);
    }
  }

  return (
    <main className="auth-page">
      <div className="auth-card">
        <div className="auth-brand">
          Firm<span>Ledger</span>
        </div>

        <h1>Sign in</h1>

        <form onSubmit={submit} className="auth-form">
          <label htmlFor="login-email">
            Email

              id="login-email"

import Link from "next/link";
import { requireRole } from "@/lib/permissions";
import { prisma } from "@/lib/prisma";

export default async function Reports() {
  const u = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const [lines, invoices, expenses, payments] = await Promise.all([
    prisma.journalLine.findMany({
      where: {
        journalEntry: {
          firmId: u.firmId,
        },
      },
      include: {
        ledgerAccount: true,
        journalEntry: true,
      },
    }),

    prisma.invoice.findMany({
      where: {
        firmId: u.firmId,
      },
      select: {
        total: true,
        amountPaid: true,
        balanceDue: true,
      },
    }),

    prisma.expense.findMany({
      where: {
        firmId: u.firmId,
        status: {
          not: "REJECTED",
        },
      },
      select: {
        amount: true,
        taxAmount: true,
      },
    }),

    prisma.payment.findMany({
      where: {
        firmId: u.firmId,
        status: "COMPLETED",
      },
      select: {
        amount: true,
      },
    }),
  ]);

  const accountTotals = new Map<
    string,
    {
      name: string;
      debit: number;
      credit: number;
      type: string;
    }
  >();

  for (const l of lines) {
    const current = accountTotals.get(l.ledgerAccountId) || {
      name: l.ledgerAccount.name,
      debit: 0,
      credit: 0,
      type: l.ledgerAccount.accountType ?? "",
    };

    current.debit += Number(l.debit);
    current.credit += Number(l.credit);

    accountTotals.set(l.ledgerAccountId, current);
  }

  const revenue = [...accountTotals.values()]
    .filter((a) => a.type === "REVENUE")
    .reduce((s, a) => s + a.credit - a.debit, 0);

  const operatingExpenses = [...accountTotals.values()]
    .filter((a) => a.type === "EXPENSE")
    .reduce((s, a) => s + a.debit - a.credit, 0);

  const receivables = invoices.reduce(
    (s, i) => s + Number(i.balanceDue),
    0
  );

  const receipts = payments.reduce(
    (s, p) => s + Number(p.amount),
    0
  );

  const expensesTotal = expenses.reduce(
    (s, e) => s + Number(e.amount) + Number(e.taxAmount),
    0
  );

  return (
    <div className="shell">
      <aside className="sidebar">
        <div className="brand">
          Firm<span>Ledger</span>
        </div>

        <Link className="nav-link" href="/">
          ← Dashboard
        </Link>

        <Link className="nav-link active" href="/reports">
          Reports
        </Link>

        <Link className="nav-link" href="/accounting">
          Accounting
        </Link>
      </aside>

      <main className="main">
        <header className="topbar">
          <strong>{u.firm.name}</strong>
        </header>

        <section className="content">
          <h1 className="page-title">Financial reports</h1>

          <p className="page-subtitle">
            Live management figures calculated from FirmLedger&apos;s posted
            transactions.
          </p>

          <div className="grid grid-4" style={{ marginTop: 24 }}>
            <div className="card">
              <div className="metric-label">Revenue</div>
              <div className="metric-value">
                R{revenue.toFixed(2)}
              </div>
            </div>

            <div className="card">
              <div className="metric-label">Operating expenses</div>
              <div className="metric-value">
                R{operatingExpenses.toFixed(2)}
              </div>
            </div>

            <div className="card">
              <div className="metric-label">Net result</div>
              <div className="metric-value">
                R{(revenue - operatingExpenses).toFixed(2)}
              </div>
            </div>

            <div className="card">
              <div className="metric-label">Accounts receivable</div>
              <div className="metric-value">
                R{receivables.toFixed(2)}
              </div>
            </div>
          </div>

          <div className="grid grid-2" style={{ marginTop: 24 }}>
            <div className="card">
              <div className="section-title">Cash receipts</div>

              <p className="page-subtitle">
                Completed client payments recorded in FirmLedger.
              </p>

              <div className="metric-value">
                R{receipts.toFixed(2)}
              </div>
            </div>

            <div className="card">
              <div className="section-title">
                Expense cash outflow
              </div>

              <p className="page-subtitle">
                Recorded paid expenses including input VAT.
              </p>

              <div className="metric-value">
                R{expensesTotal.toFixed(2)}
              </div>
            </div>
          </div>

          <div className="card" style={{ marginTop: 24 }}>
            <div className="section-title">Trial balance</div>

            <table className="table">
              <thead>
                <tr>
                  <th>Account</th>
                  <th>Type</th>
                  <th>Debit</th>
                  <th>Credit</th>
                  <th>Balance</th>
                </tr>
              </thead>

              <tbody>
                {[...accountTotals.values()]
                  .sort((a, b) =>
                    a.name.localeCompare(b.name)
                  )
                  .map((a) => {
                    const balance = a.debit - a.credit;

                    return (
                      <tr key={a.name}>
                        <td>{a.name}</td>

                        <td>{a.type}</td>

                        <td>
                          R{a.debit.toFixed(2)}
                        </td>

                        <td>
                          R{a.credit.toFixed(2)}
                        </td>

                        <td>
                          R{Math.abs(balance).toFixed(2)}{" "}
                          {balance >= 0 ? "Dr" : "Cr"}
                        </td>
                      </tr>
                    );
                  })}
              </tbody>
            </table>
          </div>
        </section>
      </main>
    </div>
  );

import { cookies } from "next/headers";
import {
  createHmac,
  randomBytes,
  pbkdf2 as pbkdf2Callback,
  timingSafeEqual,
} from "node:crypto";
import { promisify } from "node:util";
import { prisma } from "@/lib/prisma";

const pbkdf2 = promisify(pbkdf2Callback);

export const SESSION_COOKIE = "firmledger_session";
export const SESSION_TTL_SECONDS = 60 * 60 * 8;

function getAuthSecret() {
  const value = process.env.AUTH_SECRET;

  if (!value) {
    throw new Error("AUTH_SECRET is not configured.");
  }

  return value;
}

export async function hashPassword(password: string) {
  const salt = randomBytes(16).toString("hex");

  const derived = await pbkdf2(
    password,
    salt,
    120_000,
    64,
    "sha512"
  );

  return `pbkdf2$120000$${salt}$${derived.toString("hex")}`;
}

export async function verifyPassword(
  password: string,
  stored: string
) {
  try {
    const [algorithm, iterationsText, salt, expected] =
      stored.split("$");

    if (
      algorithm !== "pbkdf2" ||
      !iterationsText ||
      !salt ||
      !expected
    ) {
      return false;
    }

    const iterations = Number(iterationsText);

    if (!Number.isInteger(iterations) || iterations <= 0) {
      return false;
    }

    const derived = await pbkdf2(
      password,
      salt,
      iterations,
      64,
      "sha512"
    );

    const expectedBuffer = Buffer.from(expected, "hex");

    if (derived.length !== expectedBuffer.length) {
      return false;
    }

    return timingSafeEqual(derived, expectedBuffer);
  } catch {
    return false;
  }
}

function sign(value: string) {
  return createHmac(
    "sha256",
    getAuthSecret()
  )
    .update(value)
    .digest("hex");
}

export function makeSession(userId: string) {
  const expires =
    Math.floor(Date.now() / 1000) + SESSION_TTL_SECONDS;

  const payload = `${userId}.${expires}`;
  const signature = sign(payload);

  return `${payload}.${signature}`;
}

export function parseSession(value: string | undefined) {
  if (!value) {
    return null;
  }

  const parts = value.split(".");

  if (parts.length !== 3) {
    return null;
  }

  const [userId, expiresText, signature] = parts;

  if (!userId || !expiresText || !signature) {
    return null;
  }

  const payload = `${userId}.${expiresText}`;

  let expectedSignature: string;

  try {
    expectedSignature = sign(payload);
  } catch {
    return null;
  }

  const actualBuffer = Buffer.from(signature, "hex");
  const expectedBuffer = Buffer.from(expectedSignature, "hex");

  if (
    actualBuffer.length !== expectedBuffer.length ||
    !timingSafeEqual(actualBuffer, expectedBuffer)
  ) {
    return null;
  }

  const expires = Number(expiresText);

  if (!Number.isFinite(expires)) {
    return null;
  }

  if (expires < Math.floor(Date.now() / 1000)) {
    return null;
  }

  return {
    userId,
  };
}

export async function getCurrentUser() {
  const cookieStore = await cookies();

  const sessionCookie =
    cookieStore.get(SESSION_COOKIE)?.value;

  const session = parseSession(sessionCookie);

  if (!session) {
    return null;
  }

  return prisma.firmUser.findUnique({
    where: {
      id: session.userId,
    },
    include: {
      firm: true,
    },
  });
}

import { redirect } from "next/navigation";

export async function requireUser() {
  const user = await getCurrentUser();
  if (!user || !user.active) redirect("/login");
  return user;
}

export async function requireRole(roles: Role[]) {
  const user = await requireUser();
  if (!roles.includes(user.role)) redirect("/");
  return user;
}

  const result = await db.ledgerAccount.findFirst({ where: { firmId, code, active: true } });
  if (!result) throw new Error(`Ledger account ${code} is not configured for this firm.`);
  return result;

function assertBalanced(lines: { debit: number; credit: number }[]) {
  const debit = Math.round(lines.reduce((s,l)=>s+l.debit,0)*100);
  const credit = Math.round(lines.reduce((s,l)=>s+l.credit,0)*100);
  if (debit !== credit) throw new Error("Journal entry is not balanced.");
}
async function entry(db: Db, args: {firmId:string;date:Date;description:string;sourceType:string;sourceId:string;lines:{ledgerAccountId:string;debit:number;credit:number;description?:string}[]}) {

  return db.journalEntry.create({data:{firmId:args.firmId,entryDate:args.date,description:args.description,sourceType:args.sourceType,sourceId:args.sourceId,lines:{create:args.lines}}});
}
