export default async function Trust() {
  const u = await requireRole([
    "OWNER",
    "ADMIN",
    "FINANCE",
    "ACCOUNTANT",
  ]);

  const [accounts, tx] = await Promise.all([
    prisma.trustAccount.findMany({
      where: {
        firmId: u.firmId,
      },
      orderBy: {
        name: "asc",
      },
    }),

    prisma.trustTransaction.findMany({
      where: {
        firmId: u.firmId,
        status: "POSTED",
      },
      include: {
        client: true,
        matter: true,
        trustAccount: true,
      },
      orderBy: {
        transactionDate: "desc",
      },
      take: 50,
    }),
  ]);

  const balances = accounts.map((account) => {
    const accountTransactions = tx.filter(
      (transaction) =>
        transaction.trustAccountId === account.id
    );

    return {
      account,
      balance: accountTransactions.reduce(
        (sum, transaction) =>
          sum +
          (transaction.type === "RECEIPT" ||
          transaction.type === "INTEREST"
            ? Number(transaction.amount)
            : -Number(transaction.amount)),
        0
      ),
    };
  });

          </div>

          <div

            {balances.map((item) => (
              <div className="card" key={item.account.id}>
                <div className="metric-label">
